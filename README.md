# nvim
```
# Hark, for 'tis here Nvim's secrets oft reside
cd ~./config/nvim
# Fetch this hoard, and in this fold abide
git clone https://github.com/house-intellect/nvim.git
# Convey the font where Linux doth provide
cp ./JetBrainsMonoNerdFontMono-Regular.ttf /usr/share/fonts/
# Renew the font's cache, let all be espied
fc-cache -f -v
# Now, revel as a champion, in pride
nvim
```
